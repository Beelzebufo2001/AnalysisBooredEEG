#!/bin/bash
# =============================================================================
# CLUSTER CHEATSHEET — save this, you'll forget the commands
# =============================================================================
#
# SUBMIT A JOB
#   sbatch slurm/02_compute_corr.sbatch
#
# CHECK YOUR JOBS
#   squeue -u $USER
#
# SEE ALL NODES AND THEIR STATE
#   sinfo
#
# WATCH LIVE OUTPUT
#   tail -f logs/slurm/corr_<JOBID>_<TASKID>.out
#
# CANCEL ONE TASK (e.g. subject index 3)
#   scancel <JOBID>_3
#
# CANCEL ENTIRE ARRAY
#   scancel <JOBID>
#
# INTERACTIVE DEBUGGING SESSION ON A NODE (exit when done — it blocks resources!)
#   srun --pty --cpus-per-task=4 bash
#   then inside:
#     conda activate eeg
#     python 02_compute_corr.py --subject C01 --max-windows 5 \
#         --data-dir /data/eeg --output-dir /tmp/test_run
#     exit
#
# RUN A QUICK COMMAND DIRECTLY (non-interactive)
#   srun --cpus-per-task=8 python 02_compute_corr.py --subject C01 ...
#
# =============================================================================
# TYPICAL WORKFLOW
# =============================================================================
#
# 1. Edit SUBJECTS=(...) in each .sbatch file to match your subject list
# 2. Update #SBATCH --array=0-N where N = number of subjects - 1
# 3. Update DATA_DIR, OUTPUT_DIR paths
# 4. Generate ch_names.txt (see README)
#
# Then:
#   mkdir -p logs/slurm
#   sbatch slurm/01_plot_eeg.sbatch        # optional QC
#   sbatch slurm/02_compute_corr.sbatch    # heavy step — one job per subject
#   # wait for step 2 to finish (squeue -u $USER)
#   sbatch slurm/03_04_analyze.sbatch      # fast — all subjects in one job
#
# To make step 3 automatically wait for step 2:
#   JOB2=$(sbatch --parsable slurm/02_compute_corr.sbatch)
#   sbatch --dependency=afterok:$JOB2 slurm/03_04_analyze.sbatch
# =============================================================================
